cmake_minimum_required(VERSION 3.16)
project(REVERSE_WORKERS_C02 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(MUNDO "${ROOT}/deps/mundo-x")
if(NOT EXISTS "${ROOT}/build/ether-compat/mundo_x/quantum_runtime.hpp")
  message(FATAL_ERROR "Prepare verified original QuantumRuntime compatibility overlay first")
endif()
add_executable(reverse_workers_c02 ../reverse_workers.cpp)
target_include_directories(reverse_workers_c02 BEFORE PRIVATE
  "${ROOT}/src" "${ROOT}/build/ether-compat" "${MUNDO}/native/include")
target_compile_definitions(reverse_workers_c02 PRIVATE REVERSE_PC_HAVE_BOUNDED_PRODUCT_IMPORT=1)
target_link_libraries(reverse_workers_c02 PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(reverse_workers_c02 PRIVATE /W3)
else()
  target_compile_options(reverse_workers_c02 PRIVATE -O2 -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME reverse_rvpc2_workers_c02 COMMAND reverse_workers_c02)
set_tests_properties(reverse_rvpc2_workers_c02 PROPERTIES TIMEOUT 120)
